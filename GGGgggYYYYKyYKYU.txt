local dragging, dragStart, startPos = false, nil, nil
getgenv().isDraggingPill = false

MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        getgenv().isDraggingPill = false
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        if delta.Magnitude > 5 then getgenv().isDraggingPill = true end
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local LocalPlayer = Players.LocalPlayer

local ghostFolder = nil
local ghostPartsMapping = {}

-- Hitbox Loop
task.spawn(function()
    while task.wait(0.5) do
        if getgenv().HitboxSettings and getgenv().HitboxSettings.Enabled then
            pcall(function()
                for _, player in pairs(Players:GetPlayers()) do
                    if player ~= LocalPlayer and player.Character then
                        local humanoid = player.Character:FindFirstChild("Humanoid")
                        local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                        if humanoid and humanoid.Health > 0 and hrp then
                            hrp.Size = Vector3.new(getgenv().HitboxSettings.Size, getgenv().HitboxSettings.Size, getgenv().HitboxSettings.Size)
                            hrp.Transparency = getgenv().HitboxSettings.Transparency
                            hrp.Color = getgenv().HitboxSettings.Color
                            hrp.Material = Enum.Material.Neon
                            hrp.CanCollide = false
                        end
                    end
                end
            end)
        end
    end
end)

-- Noclip Loop
RunService.Stepped:Connect(function()
    if getgenv().NoclipSettings and getgenv().NoclipSettings.Enabled then
        pcall(function()
            if LocalPlayer.Character then
                for _, part in pairs(LocalPlayer.Character:GetDescendants()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
            end
        end)
    end
end)

-- RenderStepped Logic (Tàng hình triệt để + Ghost cho bản thân + Speed + ESP)
RunService.RenderStepped:Connect(function()
    if getgenv().InvisibleSettings then
        pcall(function()
            if LocalPlayer.Character then
                if getgenv().InvisibleSettings.Enabled then
                    -- Ẩn tuyệt đối với người khác (Độ mờ = 1 và xóa quần áo 2D/phụ kiện hiển thị texture)
                    for _, obj in pairs(LocalPlayer.Character:GetDescendants()) do
                        if obj:IsA("BasePart") or obj:IsA("Decal") then 
                            obj.Transparency = 1 
                        elseif obj:IsA("Shirt") or obj:IsA("Pants") or obj:IsA("ShirtGraphic") or obj:IsA("Accessory") then
                            obj:Destroy() -- Xóa quần áo và phụ kiện để người khác không thấy dính hình dạng
                        end
                    end
                    
                    -- Tạo bóng mờ trắng cho riêng bạn nhìn thấy qua Camera
                    local cam = workspace.CurrentCamera
                    if cam then
                        if not ghostFolder or not ghostFolder.Parent then
                            if ghostFolder then ghostFolder:Destroy() end
                            ghostFolder = Instance.new("Folder")
                            ghostFolder.Name = "LocalGhostCharacter"
                            ghostFolder.Parent = cam
                            ghostPartsMapping = {}
                            for _, obj in pairs(LocalPlayer.Character:GetDescendants()) do
                                if obj:IsA("BasePart") and obj.Name ~= "HumanoidRootPart" then
                                    local clone = obj:Clone()
                                    clone.Parent = ghostFolder
                                    for _, c in pairs(clone:GetChildren()) do
                                        if not c:IsA("SpecialMesh") and not c:IsA("FileMesh") then c:Destroy() end
                                    end
                                    clone.Color = Color3.fromRGB(255, 255, 255)
                                    clone.Material = Enum.Material.Neon
                                    clone.Transparency = 0.5
                                    clone.CanCollide = false
                                    clone.Anchored = true
                                    ghostPartsMapping[obj] = clone
                                end
                            end
                        end
                        for realPart, ghostPart in pairs(ghostPartsMapping) do
                            if realPart and realPart.Parent and ghostPart and ghostPart.Parent then
                                ghostPart.CFrame = cam.CFrame:Inverse() * realPart.CFrame
                            else
                                ghostPartsMapping[realPart] = nil
                                if ghostPart then ghostPart:Destroy() end
                            end
                        end
                    end
                else
                    if ghostFolder then
                        ghostFolder:Destroy()
                        ghostFolder = nil
                        ghostPartsMapping = {}
                    end
                end
            end
        end)
    end

    if getgenv().SpeedSettings and getgenv().SpeedSettings.Enabled then
        pcall(function()
            if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
                local hum = LocalPlayer.Character.Humanoid
                if hum.Health > 0 then hum.WalkSpeed = getgenv().SpeedSettings.Value end
            end
        end)
    end

    if getgenv().ESPSettings and getgenv().ESPSettings.Enabled then
        pcall(function()
            for _, player in pairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local char = player.Character
                    local hum = char:FindFirstChild("Humanoid")
                    local hrp = char:FindFirstChild("HumanoidRootPart")
                    if hum and hum.Health > 0 and hrp then
                        local highlight = char:FindFirstChild("ESPHighlight")
                        if not highlight then
                            highlight = Instance.new("Highlight")
                            highlight.Name = "ESPHighlight"
                            highlight.FillTransparency = 0.6
                            highlight.FillColor = getgenv().ESPSettings.BoxColor
                            highlight.OutlineColor = getgenv().ESPSettings.BoxColor
                            highlight.Parent = char
                        end
                        local espText = hrp:FindFirstChild("ESPText")
                        if not espText then
                            espText = Instance.new("BillboardGui")
                            espText.Name = "ESPText"
                            espText.Size = UDim2.new(0, 200, 0, 50)
                            espText.StudsOffset = Vector3.new(0, 3, 0)
                            espText.AlwaysOnTop = true
                            local label = Instance.new("TextLabel")
                            label.Size = UDim2.new(1, 0, 1, 0)
                            label.BackgroundTransparency = 1
                            label.TextColor3 = getgenv().ESPSettings.TextColor
                            label.TextStrokeTransparency = 0
                            label.Font = Enum.Font.GothamBold
                            label.TextSize = 13
                            label.Parent = espText
                            espText.Parent = hrp
                        end
                        local action = "Đứng yên"
                        if hum.Sit then action = "Đang ngồi"
                        elseif hum:GetState() == Enum.HumanoidStateType.Freefall or hum:GetState() == Enum.HumanoidStateType.Jumping then action = "Đang nhảy/rơi"
                        elseif hum.MoveDirection.Magnitude > 0 then action = "Đang di chuyển" end
                        local tool = char:FindFirstChildOfClass("Tool")
                        local toolName = tool and tool.Name or "Tay không"
                        espText.TextLabel.Text = string.format("[%s]\n%s\nCầm: %s", player.Name, action, toolName)
                    else
                        if char:FindFirstChild("ESPHighlight") then char.ESPHighlight:Destroy() end
                        if hrp and hrp:FindFirstChild("ESPText") then hrp.ESPText:Destroy() end
                    end
                end
            end
        end)
    end
end)

Players.PlayerRemoving:Connect(function(player)

        local hl = player.Character:FindFirstChild("ESPHighlight")
        if hl then hl:Destroy() end
    end
end)
