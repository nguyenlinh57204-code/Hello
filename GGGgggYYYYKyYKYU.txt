-- ================= HỆ THỐNG KÉO THẢ (DRAG SYSTEM) ================= --
local dragging = false
local dragStart, startPos
local isDraggingPill = false

MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        isDraggingPill = false
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        if delta.Magnitude > 5 then
            isDraggingPill = true
        end
        
        MainFrame.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

-- ================= CÁC HÀM XỬ LÝ ================= --

local function ResetHitboxes()
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = player.Character.HumanoidRootPart
            hrp.Size = Vector3.new(2, 2, 1)
            hrp.Transparency = 1
            hrp.Material = Enum.Material.Plastic
        end
    end
end

local function ClearESP()
    for _, player in pairs(Players:GetPlayers()) do
        if player.Character then
            local hl = player.Character:FindFirstChild("ESPHighlight")
            if hl then hl:Destroy() end
            
            local hrp = player.Character:FindFirstChild("HumanoidRootPart")
            if hrp and hrp:FindFirstChild("ESPText") then
                hrp.ESPText:Destroy()
            end
        end
    end
end

-- Logic Tàng Hình (Ẩn với người khác, tạo hiệu ứng bóng trắng mờ cho riêng mình)
ToggleInvisibleBtn.MouseButton1Click:Connect(function()
    InvisibleSettings.Enabled = not InvisibleSettings.Enabled
    if InvisibleSettings.Enabled then
        ToggleInvisibleBtn.Text = "Tàng hình: ĐANG BẬT"
        ToggleInvisibleBtn.TextColor3 = Color3.fromRGB(100, 255, 100)
        
        ghostPartsMapping = {}
        if ghostFolder then ghostFolder:Destroy() end
        
        -- Tạo thư mục chứa nhân vật giả lập chỉ hiển thị trên máy của bạn (qua Camera)
        ghostFolder = Instance.new("Folder")
        ghostFolder.Name = "LocalGhostCharacter"
        ghostFolder.Parent = workspace.CurrentCamera
        
        if LocalPlayer.Character then
            for _, obj in pairs(LocalPlayer.Character:GetDescendants()) do
                if obj:IsA("BasePart") and obj.Name ~= "HumanoidRootPart" then
                    local clone = obj:Clone()
                    clone.Parent = ghostFolder
                    for _, child in pairs(clone:GetChildren()) do
                        if not child:IsA("SpecialMesh") and not child:IsA("FileMesh") then
                            child:Destroy()
                        end
                    end
                    clone.Color = Color3.fromRGB(255, 255, 255) -- Màu trắng mờ
                    clone.Material = Enum.Material.Neon -- Phát sáng nhẹ giúp dễ nhìn
                    clone.Transparency = 0.5 -- Mờ trong suốt
                    clone.CanCollide = false
                    clone.Anchored = true
                    
                    ghostPartsMapping[obj] = clone
                end
            end
        end
    else
        ToggleInvisibleBtn.Text = "Tàng hình: TẮT"
        ToggleInvisibleBtn.TextColor3 = Color3.fromRGB(255, 100, 100)
        
        if ghostFolder then
            ghostFolder:Destroy()
            ghostFolder = nil
        end
        ghostPartsMapping = {}
        
        if LocalPlayer.Character then
            for _, obj in pairs(LocalPlayer.Character:GetDescendants()) do
                if obj:IsA("BasePart") or obj:IsA("Decal") then
                    obj.Transparency = 0
                end
            end
        end
    end
end)

-- ================= VÒNG LẶP CHÍNH ================= --

-- Vòng lặp Hitbox (0.5 giây/lần)
task.spawn(function()
    while task.wait(0.5) do
        if HitboxSettings.Enabled then
            pcall(function()
                for _, player in pairs(Players:GetPlayers()) do
                    if player ~= LocalPlayer and player.Character then
                        local humanoid = player.Character:FindFirstChild("Humanoid")
                        local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                        
                        if humanoid and humanoid.Health > 0 and hrp then
                            hrp.Size = Vector3.new(HitboxSettings.Size, HitboxSettings.Size, HitboxSettings.Size)
                            pcall(function()
                                hrp.Transparency = HitboxSettings.Transparency
                                hrp.Color = HitboxSettings.Color
                                hrp.Material = Enum.Material.Neon
                            end)
                            hrp.CanCollide = false
                        end
                    end
                end
            end)
        end
    end
end)

-- Vòng lặp Noclip (Đi xuyên tường)
RunService.Stepped:Connect(function()
    if NoclipSettings.Enabled then
        pcall(function()
            if LocalPlayer.Character then
                for _, part in pairs(LocalPlayer.Character:GetDescendants()) do
                    if part:IsA("BasePart") then
                        part.CanCollide = false
                    end
                end
            end
        end)
    end
end)

-- Vòng lặp RenderStepped: ESP, Speed và Tàng Hình
RunService.RenderStepped:Connect(function()
    -- Logic Tàng Hình (Ẩn nhân vật thật với người khác, cập nhật bóng nhân vật giả cho bạn)
    if InvisibleSettings.Enabled then
        pcall(function()
            if LocalPlayer.Character then
                for _, obj in pairs(LocalPlayer.Character:GetDescendants()) do
                    if obj:IsA("BasePart") or obj:IsA("Decal") then
                        obj.Transparency = 1
                    end
                end
                
                local cam = workspace.CurrentCamera
                if cam then
                    for realPart, ghostPart in pairs(ghostPartsMapping) do
                        if realPart and realPart.Parent and ghostPart and ghostPart.Parent then
                            ghostPart.CFrame = cam.CFrame:Inverse() * realPart.CFrame
                        else
                            ghostPartsMapping[realPart] = nil
                            if ghostPart then ghostPart:Destroy() end
                        end
                    end
                end
            end
        end)
    end

    -- Logic Speed
    if SpeedSettings.Enabled then
        pcall(function()
            if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
                local hum = LocalPlayer.Character.Humanoid
                if hum.Health > 0 then
                    hum.WalkSpeed = SpeedSettings.Value
                end
            end
        end)
    end

    -- Logic ESP
    if ESPSettings.Enabled then
        pcall(function()
            for _, player in pairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local char = player.Character
                    local hum = char:FindFirstChild("Humanoid")
                    local hrp = char:FindFirstChild("HumanoidRootPart")
                    
                    if hum and hum.Health > 0 and hrp then
                        local highlight = char:FindFirstChild("ESPHighlight")
                        if not highlight then
                            highlight = Instance.new("Highlight")
                            highlight.Name = "ESPHighlight"
                            highlight.FillTransparency = 0.6
                            highlight.FillColor = ESPSettings.BoxColor
                            highlight.OutlineColor = ESPSettings.BoxColor
                            highlight.Parent = char
                        end
                        
                        local espText = hrp:FindFirstChild("ESPText")
                        if not espText then
                            espText = Instance.new("BillboardGui")
                            espText.Name = "ESPText"
                            espText.Size = UDim2.new(0, 200, 0, 50)
                            espText.StudsOffset = Vector3.new(0, 3, 0)
                            espText.AlwaysOnTop = true
                            
                            local label = Instance.new("TextLabel")
                            label.Size = UDim2.new(1, 0, 1, 0)
                            label.BackgroundTransparency = 1
                            label.TextColor3 = ESPSettings.TextColor
                            label.TextStrokeTransparency = 0
                            label.Font = Enum.Font.GothamBold
                            label.TextSize = 13
                            label.Parent = espText
                            
                            espText.Parent = hrp
                        end
                        
                        local action = "Đứng yên"
                        if hum.Sit then
                            action = "Đang ngồi"
                        elseif hum:GetState() == Enum.HumanoidStateType.Freefall or hum:GetState() == Enum.HumanoidStateType.Jumping then
                            action = "Đang nhảy/rơi"
                        elseif hum.MoveDirection.Magnitude > 0 then
                            action = "Đang di chuyển"
                        end
                        
                        local tool = char:FindFirstChildOfClass("Tool")
                        local toolName = tool and tool.Name or "Tay không"
                        
                        espText.TextLabel.Text = string.format("[%s]\n%s\nCầm: %s", player.Name, action, toolName)
                    else
                        if char:FindFirstChild("ESPHighlight") then char.ESPHighlight:Destroy() end
                        if hrp and hrp:FindFirstChild("ESPText") then hrp.ESPText:Destroy() end
                    end
                end
            end
        end)
    end
end)
